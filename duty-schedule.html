    function switchTab(tab) {

      document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));

      document.querySelectorAll('.panel').forEach(p => p.classList.remove('active'));

      event.target.classList.add('active');

      document.getElementById(tab).classList.add('active');

    }


    function filterNames() {

      const q = document.getElementById('nameSearch').value.trim();

      document.querySelectorAll('.name-chip').forEach(chip => {

        chip.style.display = chip.textContent.includes(q) ? '' : 'none';

      });

    }


    function selectName(name, chipEl) {

      document.querySelectorAll('.name-chip').forEach(c => c.classList.remove('selected'));

      chipEl.classList.add('selected');


      const rows = [];

      schedule.forEach(row => {

        const matched = duties.filter(d => row[d.key] === name);

        if (matched.length > 0) {

          rows.push({ day: row.day, duties: matched });

        }

      });


      const resultDiv = document.getElementById('nameResult');

      if (rows.length === 0) {

        resultDiv.innerHTML = '<div class="empty-state">ไม่พบเวรสำหรับ ' + name + '</div>';

        return;

      }


      let html = '<div class="result-card"><h3>เวรของ ' + name + ' (สิงหาคม 2026)</h3>';

      html += '<table class="result-table"><thead><tr><th>วันที่</th><th>เวร</th></tr></thead><tbody>';

      rows.forEach(r => {

        html += '<tr><td><span class="day-badge">' + r.day + '</span></td><td>';

        r.duties.forEach(d => {

          html += '<span class="duty-tag ' + d.cls + '">' + d.label + '</span>';

        });

        html += '</td></tr>';

      });

      html += '</tbody></table></div>';

      resultDiv.innerHTML = html;

    }

    const dayHeaders = ['อา','จ','อ','พ','พฤ','ศ','ส'];

    const firstDayOfWeek = 6; // Aug 1, 2026 = Saturday (0=Sun)


    function renderCalendar() {

      const grid = document.getElementById('calGrid');

      dayHeaders.forEach(d => {

        const h = document.createElement('div');

        h.className = 'cal-day-header';

        h.textContent = d;

        grid.appendChild(h);

      });


      // Empty cells before day 1

      for (let i = 0; i < firstDayOfWeek; i++) {

        empty.className = 'cal-day empty';

        grid.appendChild(empty);

      }


      schedule.forEach(row => {

        const cell = document.createElement('div');

        cell.className = 'cal-day';

        cell.onclick = () => openDayModal(row.day);


        let dutiesHtml = '<div class="dnum">' + row.day + '</div><div class="duties">';

        duties.forEach(d => {

          const v = row[d.key];

          }

        });

        dutiesHtml += '</div>';

        cell.innerHTML = dutiesHtml;

        grid.appendChild(cell);

      });

    }


    // ===== MODAL =====

    function openDayModal(day) {

      const row = schedule.find(r => r.day === day);

      if (!row) return;


      document.getElementById('modalTitle').textContent = 'เวรวันที่ ' + day + ' สิงหาคม 2026';


      let html = '';

      duties.forEach(d => {

        const v = row[d.key];

        if (v && v !== '.') {

          html += '<div class="duty-row"><span class="duty-tag ' + d.cls + '">' + d.label + '</span><span class="dname">' + v + '</span></div>';

        }

      });

      if (!html) html = '<div class="empty-state">ไม่มีเวรในวันนี้</div>';


      document.getElementById('modalContent').innerHTML = html;

      document.getElementById('dayModal').classList.add('show');

    }


    function closeModal(e) {

      if (!e || e.target === document.getElementById('dayModal')) {

        document.getElementById('dayModal').classList.remove('show');

      }

    }
